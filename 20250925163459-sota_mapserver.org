* Overview
Parameterized, repeatable pipeline to publish SOTA summits through GeoServer
(WFS 2.0.0) and consume in CalTopo. Uses a persistent bash session and noweb
references for clean, reusable code blocks.

All shell blocks use =:session sota= for variable persistence.

* Fetch SOTA CSV and Clean Header
The first line in the SOTA CSV is a title, not headers.

#+begin_src sh
curl -fsSL -o "$DATA_DIR/summitslist.csv" https://storage.sota.org.uk/summitslist.csv

# Remove title line (keep header and all data)
tail -n +2 "$DATA_DIR/summitslist.csv" > "$DATA_DIR/summitslist_clean.csv"

echo "Line counts:"
wc -l "$DATA_DIR"/summitslist*.csv
echo ""
echo "First 2 lines of cleaned CSV:"
head -n 2 "$DATA_DIR/summitslist_clean.csv"
#+end_src

* Build GeoPackage (CSV → GPKG)
Convert CSV to GeoPackage with proper geometry and CRS:
- Geometry from =Longitude= and =Latitude= columns
- Geometry column name: =the_geom=
- Adds computed SOTLAS field (https://sotl.as/summits/ + SummitCode)

#+begin_src sh
# Step 1: Create basic GPKG with geometry from CSV
ogr2ogr -f GPKG "$GPKG" \
  "$DATA_DIR/summitslist_clean.csv" \
  -oo HEADERS=YES -oo AUTODETECT_TYPE=YES \
  -oo X_POSSIBLE_NAMES=Longitude -oo Y_POSSIBLE_NAMES=Latitude \
  -nln "${LAYER}_temp" \
  -nlt POINT -a_srs EPSG:4326 -lco GEOMETRY_NAME=the_geom


# Step 2: Add SOTLAS field and
# convert ActivationCount to text so even "0" will be displayed
ogr2ogr -f GPKG "$GPKG" "$GPKG" "${LAYER}_temp" \
  -nln "$LAYER" \
  -sql "SELECT *, 
        'https://sotl.as/summits/' || SummitCode AS SOTLAS,
        CAST(ActivationCount AS TEXT) AS Activations,
        the_geom 
        FROM ${LAYER}_temp" \
  -dialect SQLite \
  -update -overwrite

# Step 3: Remove temporary layer
ogrinfo "$GPKG" -sql "DROP TABLE ${LAYER}_temp"

echo ""
echo "Sanity checks:"
ogrinfo "$GPKG" -sql "SELECT COUNT(*) AS total FROM $LAYER"
echo ""
echo "Sample features with geometry and SOTLAS:"
ogrinfo "$GPKG" -sql "SELECT SummitCode, SummitName, SOTLAS, Activations, AsText(the_geom) AS wkt FROM $LAYER LIMIT 3"
#+end_src

* GeoServer Configuration (Automated via REST API)

Wait for GeoServer to be fully started before running these blocks.

** Check GeoServer Status
Verify GeoServer is ready to accept REST API calls:

#+begin_src sh
echo "Checking GeoServer status..."
echo "Container status:"
docker ps --filter "name=geoserver" --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}"
echo ""

echo "Testing REST API endpoint..."
RESPONSE=$(curl -s -w "\n%{http_code}" "$GS_BASE/rest/about/version.json" -u admin:geoserver)
HTTP_CODE=$(echo "$RESPONSE" | tail -n 1)

if [ "$HTTP_CODE" = "200" ]; then
  echo "✓ GeoServer is ready (HTTP $HTTP_CODE)"
  echo "$RESPONSE" | sed '$d' | jq -M '.about.resource[0] | {name, version}'
else
  echo "✗ GeoServer not ready yet (HTTP $HTTP_CODE)"
  echo "Wait 30-60 seconds and try again, or check logs with:"
  echo "  docker compose -f $ROOT_DIR/docker-compose.yml logs geoserver"
fi
echo ""
#+end_src

** Create Workspace
#+begin_src sh
echo "Creating workspace: $WS"
RESPONSE=$(curl -s -w "\n%{http_code}" -X POST "$GS_BASE/rest/workspaces" \
  -H "Content-Type: application/json" \
  -u admin:geoserver \
  -d "{\"workspace\":{\"name\":\"$WS\"}}")

HTTP_CODE=$(echo "$RESPONSE" | tail -n 1)
BODY=$(echo "$RESPONSE" | sed '$d')

if [ "$HTTP_CODE" = "201" ]; then
  echo "✓ Workspace created successfully (HTTP $HTTP_CODE)"
elif [ "$HTTP_CODE" = "409" ] || [ "$HTTP_CODE" = "401" ] || ([ "$HTTP_CODE" = "500" ] && echo "$BODY" | grep -q "already exists"); then
  echo "⚠ Workspace already exists - continuing..."
else
  echo "✗ Error creating workspace (HTTP $HTTP_CODE)"
  echo "Response: $BODY"
fi
echo ""
#+end_src

** Create GeoPackage Datastore
#+begin_src sh

HTTP_CODE=$(echo "$RESPONSE" | tail -n 1)

if [ "$HTTP_CODE" = "201" ]; then

** Publish Layer
#+begin_src sh
echo "Publishing layer: $LAYER"
RESPONSE=$(curl -s -w "\n%{http_code}" -X POST "$GS_BASE/rest/workspaces/$WS/datastores/sota_gpkg/featuretypes" \
  -H "Content-Type: application/json" \
  -u admin:geoserver \
  -d "{
    \"featureType\": {
      \"name\": \"$LAYER\",
      \"nativeName\": \"$LAYER\",
      \"title\": \"SOTA Summits\",
      \"abstract\": \"Summits on the Air (SOTA) summit locations\",
      \"enabled\": true,
      \"srs\": \"EPSG:4326\",
      \"projectionPolicy\": \"FORCE_DECLARED\"
    }
  }")

HTTP_CODE=$(echo "$RESPONSE" | tail -n 1)
BODY=$(echo "$RESPONSE" | sed '$d')

if [ "$HTTP_CODE" = "201" ]; then
  echo "✓ Layer published successfully (HTTP $HTTP_CODE)"
elif [ "$HTTP_CODE" = "500" ] && echo "$BODY" | grep -q "already exists"; then
  echo "⚠ Layer already exists (continuing...)"
else
  echo "✗ Error publishing layer (HTTP $HTTP_CODE)"
  if [ -n "$BODY" ]; then
    echo "Response: $BODY"
  else
    echo "No response body returned"
    echo ""
    echo "Checking what layers are available to publish:"
    curl -s "$GS_BASE/rest/workspaces/$WS/datastores/sota_gpkg/featuretypes.json?list=available" \
      -u admin:geoserver | jq -M -r '.list.string[]? // "No layers found in GPKG"'
  fi
fi
echo ""
#+end_src

** Configure Bounding Boxes
#+begin_src sh
echo "Configuring bounding boxes for layer: $LAYER"

# GeoServer requires a body for PUT requests, even when using recalculate
echo "Recalculating bounding boxes from data..."
RESPONSE=$(curl -s -w "\n%{http_code}" -X PUT \
  "$GS_BASE/rest/workspaces/$WS/datastores/sota_gpkg/featuretypes/$LAYER.json?recalculate=nativebbox,latlonbbox" \
  -H "Content-Type: application/json" \
  -u admin:geoserver \
  -d '{"featureType":{}}')

HTTP_CODE=$(echo "$RESPONSE" | tail -n 1)
BODY=$(echo "$RESPONSE" | sed '$d')

if [ "$HTTP_CODE" = "200" ]; then
  echo "✓ Bounding boxes recalculated successfully"
else
  echo "⚠ Warning: Bounding box recalculation returned HTTP $HTTP_CODE"
  if [ -n "$BODY" ]; then
    echo "Response: $BODY"
  fi
fi
echo ""
#+end_src

** Verify Configuration
#+begin_src sh
echo "Verifying workspace and layer configuration..."
echo ""
echo "=== Workspace ==="
curl -s "$GS_BASE/rest/workspaces/$WS.json" -u admin:geoserver | jq -M '.'
echo ""
echo "=== Datastore ==="
curl -s "$GS_BASE/rest/workspaces/$WS/datastores/sota_gpkg.json" -u admin:geoserver | jq -M '.dataStore | {name, type, enabled}'

echo "=== Layer (checking if it exists) ==="
LAYER_RESPONSE=$(curl -s -w "\n%{http_code}" "$GS_BASE/rest/workspaces/$WS/datastores/sota_gpkg/featuretypes/$LAYER.json" -u admin:geoserver)
HTTP_CODE=$(echo "$LAYER_RESPONSE" | tail -n 1)
LAYER_JSON=$(echo "$LAYER_RESPONSE" | sed '$d')

  echo "Currently published layers:"
  PUBLISHED=$(curl -s "$GS_BASE/rest/workspaces/$WS/datastores/sota_gpkg/featuretypes.json" -u admin:geoserver)
  echo "$PUBLISHED" | jq -M -r '.featureTypes.featureType[]?.name // "None published"'

* GeoServer Configuration (Manual UI Steps - Alternative)
If you prefer to configure via the UI instead of using the REST API above:

1. Login at http://localhost:8080/geoserver with =admin= / =geoserver=
2. *Workspaces* → Create new workspace =sota= (if not exists)
3. *Stores → Add new Store* → GeoPackage
   - Workspace: =sota=
   - Data Source Name: =sota_gpkg=
   - Database: browse to =$DATA_DIR/sota.gpkg=
   - Save
4. *Layers → Add a new resource*
   - Choose =sota:SOTA_Summits= from the store
   - Click "Publish"
5. On the layer configuration page:
   - *Compute from data* (Native Bounding Box)
   - *Compute from lat/lon bounds* (Lat/Lon Bounding Box)
   - Save

* Troubleshooting

** Common Issues

*** Empty or Zero Bounding Box
If "Compute from data" shows bbox near zero:
- Verify the GPKG has non-null =the_geom= column
- Re-run the "Build GeoPackage" step
- Check with: =ogrinfo "$GPKG" -sql "SELECT the_geom FROM $LAYER LIMIT 5"=

*** WFS Requests Hang
Restart the container:
#+begin_src sh
docker compose -f "$ROOT_DIR/docker-compose.yml" restart geoserver
sleep 5
docker ps --filter "name=geoserver"
#+end_src

*** CalTopo Connection Errors
- Ensure you're using the HTTPS ngrok URL (not HTTP localhost)
- Verify ngrok is running: =curl -I <ngrok-url>=
- Check GeoServer logs for errors

*** Attribute Not Found
- Check layer fields in GeoServer UI
- Test with: =ogrinfo "$GPKG" -sql "SELECT * FROM $LAYER LIMIT 1"=

* Reset Pipeline

⚠️ **WARNING:** This will remove all project data and Docker resources!

This will:
- Stop and remove the GeoServer container
- Remove all GeoServer Docker images
- Delete the entire ROOT_DIR (including all data files)
- Keep installed software (apt packages)

#+begin_src sh
echo "=== COMPLETE RESET ==="
echo ""
echo "⚠️  This will remove:"
echo "  - GeoServer Docker container and images"
echo "  - All data in $ROOT_DIR"
echo ""

# Stop and remove container
echo "Stopping GeoServer container..."
docker stop geoserver 2>/dev/null || echo "  Container not running"
docker rm geoserver 2>/dev/null || echo "  Container already removed"

# Remove all GeoServer images
echo ""
echo "Removing GeoServer Docker images..."
GEOSERVER_IMAGES=$(docker images kartoza/geoserver -q)
if [ -n "$GEOSERVER_IMAGES" ]; then
  docker rmi $GEOSERVER_IMAGES 2>/dev/null && echo "  ✓ Removed GeoServer images" || echo "  Images already removed"
else
  echo "  No GeoServer images found"
fi

# Remove project directory
echo ""
echo "Removing project directory: $ROOT_DIR"
if [ -d "$ROOT_DIR" ]; then
  rm -rf "$ROOT_DIR"
  echo "  ✓ Removed $ROOT_DIR"
else
  echo "  Directory does not exist"
fi

echo ""
echo "Cleanup complete!"
echo ""
echo "Installed software (gdal-bin, docker, ngrok, jq) is still available."
echo "To rebuild from scratch, run the org file from 'Initialize Session'."
#+end_src
